+   Sale solo lo público del perfil (@, nombre e inicial del apellido, bio, marca favorita, foto), los conteos (catas públicas y sellos por tipo)

+   Nunca sale el humidor, los favoritos, las estrellas sueltas, cuáles lugares selló, la llave de una foto del teléfono, el id de una entrada,

+import { checkHandle, checkAvatar, publicName } from './profile.js';

+export const KINDS = ['artesanos', 'cigarros', 'lugares'];
+export const PUBLIC_MAX = 100; // catas públicas que se muestran; el conteo es de todas
+const MAX = { nombre: 80, marca: 60, artesano: 80, notas: 1000, tag: 40, tags: 30 };
+const DAY = /^\d{4}-\d{2}-\d{2}$/;
+const isObj = o => o !== null && typeof o === 'object' && !Array.isArray(o);
+
+// texto recortado por caracteres, no por bytes; lo que no es texto queda vacío
+const str = (v, n) => typeof v === 'string' ? [...v.trim()].slice(0, n).join('') : '';
+
+// una cata pública con solo lo que se enseña: lo demás de la entrada (foto, id, fecha de guardado, pasos) se queda en su Mi Hoja
+export function publicEntry(e) {
+  const r = e.rating;
+  return {
+    nombre: str(e.nombre, MAX.nombre),
+    marca: str(e.marca, MAX.marca),
+    artesano: str(e.artesano, MAX.artesano),
+    rating: Number.isInteger(r) && r >= 1 && r <= 5 ? r : 0,
+    tags: (Array.isArray(e.tags) ? e.tags : []).map(t => str(t, MAX.tag)).filter(Boolean).slice(0, MAX.tags),
+    notas: str(e.notas, MAX.notas),
+    consumido: typeof e.consumido === 'string' && DAY.test(e.consumido) ? e.consumido : null
+  };
+}
+
+// del documento guardado: las catas públicas (la más reciente primero, por el día en que se fumó) y los sellos por tipo
+export function publicDoc(text) {
+  let doc = null;
+  try { doc = text == null ? null : JSON.parse(text); } catch (e) { doc = null; }
+  if (!isObj(doc)) doc = {};
+  const all = (Array.isArray(doc.diario) ? doc.diario : []).filter(e => isObj(e) && e.publica === true).map(publicEntry);
+  // orden estable: a igual día se queda el orden del diario (la más nueva primero)
+  const catas = all.map((e, i) => [e, i]).sort((a, b) => (b[0].consumido || '').localeCompare(a[0].consumido || '') || a[1] - b[1]).map(x => x[0]);
+  const pas = isObj(doc.pasaporte) ? doc.pasaporte : {};
+  const sellos = {};
+  for (const k of KINDS) sellos[k] = isObj(pas[k]) ? Object.keys(pas[k]).length : 0;
+  return { catas, sellos };
+}
+
+// { uid, view } o null. uid es solo para quien llama (los conteos de seguidores en LHB-58) y nunca se manda; view es lo que se puede enseñar
+export async function publicPassport(db, rawHandle) {
+  if (typeof rawHandle !== 'string') return null;
+  const h = checkHandle(rawHandle);
+  if (h.error) return null;
+  const row = await db.prepare(
+    'SELECT p.user_id, p.handle, p.first_name, p.last_name, p.bio, p.favorite_brand, p.avatar_key, m.doc FROM profiles p ' +
+    'LEFT JOIN mihoja m ON m.user_id = p.user_id WHERE p.handle = ?1 AND p.hidden = 0'
+  ).bind(h.handle).first();
+  if (!row) return null;
+  const { catas, sellos } = publicDoc(row.doc);
+  const av = checkAvatar(row.avatar_key);
+  return {
+    uid: row.user_id,
+    view: {
+      profile: {
+        handle: row.handle,
+        display_name: row.first_name && row.last_name ? publicName(row.first_name, row.last_name) : null,
+        bio: row.bio || null,
+        favorite_brand: row.favorite_brand || null,
+        avatar_key: av.error ? null : av.avatar_key
+      },
+      counts: { catas: catas.length, sellos },
+      catas: catas.slice(0, PUBLIC_MAX)
+    }
+  };
+}

export function checkBirthDate(v, today = utcDay()) {
  if (v == null || v === '') return { error: 'birth_date-required' };
  if (typeof v !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(v)) return { error: 'birth_date-invalid' };
  const y = +v.slice(0, 4), m = +v.slice(5, 7), d = +v.slice(8, 10);
  const dt = new Date(0);

+/* LHB-40: el pasaporte público de una persona (/u/<handle>), una sola lectura que también usa la app (LHB-58).
+   Sale solo lo público del perfil (@, nombre e inicial del apellido, bio, marca favorita, foto), los conteos (catas públicas y sellos por tipo)
+   y las catas que su dueño marcó públicas (publica === true), cada una con una lista cerrada de campos.
+   Nunca sale el humidor, los favoritos, las estrellas sueltas, cuáles lugares selló, la llave de una foto del teléfono, el id de una entrada,
+   el apellido completo, el género ni la fecha de nacimiento. Un perfil oculto o que no existe es null (los dos dan 404 igual). */
+import { checkHandle, checkAvatar, publicName } from './profile.js';
+
+export const KINDS = ['artesanos', 'cigarros', 'lugares'];
+export const PUBLIC_MAX = 100; // catas públicas que se muestran; el conteo es de todas
+const MAX = { nombre: 80, marca: 60, artesano: 80, notas: 1000, tag: 40, tags: 30 };
+const DAY = /^\d{4}-\d{2}-\d{2}$/;
+const isObj = o => o !== null && typeof o === 'object' && !Array.isArray(o);
+
+// texto recortado por caracteres, no por bytes; lo que no es texto queda vacío
+const str = (v, n) => typeof v === 'string' ? [...v.trim()].slice(0, n).join('') : '';
+
+// una cata pública con solo lo que se enseña: lo demás de la entrada (foto, id, fecha de guardado, pasos) se queda en su Mi Hoja
+export function publicEntry(e) {
+  const r = e.rating;
+  return {
+    nombre: str(e.nombre, MAX.nombre),
+    marca: str(e.marca, MAX.marca),
+    artesano: str(e.artesano, MAX.artesano),
+    rating: Number.isInteger(r) && r >= 1 && r <= 5 ? r : 0,
+    tags: (Array.isArray(e.tags) ? e.tags : []).map(t => str(t, MAX.tag)).filter(Boolean).slice(0, MAX.tags),
+    notas: str(e.notas, MAX.notas),
+    consumido: typeof e.consumido === 'string' && DAY.test(e.consumido) ? e.consumido : null
+  };
+}
+
+// del documento guardado: las catas públicas (la más reciente primero, por el día en que se fumó) y los sellos por tipo
+export function publicDoc(text) {
+  let doc = null;
+  try { doc = text == null ? null : JSON.parse(text); } catch (e) { doc = null; }
+  if (!isObj(doc)) doc = {};
+  const all = (Array.isArray(doc.diario) ? doc.diario : []).filter(e => isObj(e) && e.publica === true).map(publicEntry);
+  // orden estable: a igual día se queda el orden del diario (la más nueva primero)
+  const catas = all.map((e, i) => [e, i]).sort((a, b) => (b[0].consumido || '').localeCompare(a[0].consumido || '') || a[1] - b[1]).map(x => x[0]);
+  const pas = isObj(doc.pasaporte) ? doc.pasaporte : {};
+  const sellos = {};
+  for (const k of KINDS) sellos[k] = isObj(pas[k]) ? Object.keys(pas[k]).length : 0;
+  return { catas, sellos };
+}
+
+// { uid, view } o null. uid es solo para quien llama (los conteos de seguidores en LHB-58) y nunca se manda; view es lo que se puede enseñar
+export async function publicPassport(db, rawHandle) {
+  if (typeof rawHandle !== 'string') return null;
+  const h = checkHandle(rawHandle);
+  if (h.error) return null;
+  const row = await db.prepare(
+    'SELECT p.user_id, p.handle, p.first_name, p.last_name, p.bio, p.favorite_brand, p.avatar_key, m.doc FROM profiles p ' +
+    'LEFT JOIN mihoja m ON m.user_id = p.user_id WHERE p.handle = ?1 AND p.hidden = 0'
+  ).bind(h.handle).first();
+  if (!row) return null;
+  const { catas, sellos } = publicDoc(row.doc);
+  const av = checkAvatar(row.avatar_key);
+  return {
+    uid: row.user_id,
+    view: {
+      profile: {
+        handle: row.handle,
+        display_name: row.first_name && row.last_name ? publicName(row.first_name, row.last_name) : null,
+        bio: row.bio || null,
+        favorite_brand: row.favorite_brand || null,
+        avatar_key: av.error ? null : av.avatar_key
+      },
+      counts: { catas: catas.length, sellos },
+      catas: catas.slice(0, PUBLIC_MAX)
+    }
+  };
+}

# La Hoja Boricua

App web (PWA) sobre el tabaco artesanal puertorriqueño: artesanos, cigarros hechos en Puerto Rico, lugares, eventos y una sección personal ("Mi Hoja") con humidor, diario de catas, pasaporte y favoritos. Funciona en el teléfono, se puede instalar en la pantalla de inicio y sigue funcionando sin conexión.

**En vivo:** https://lahojaboricua.com

> Prototipo: información pública, sujeta a verificación.

## Estructura

- `data/`: los datos fuente en JSON (`artesanos.json`, `cigarros.json`, `places.json`, `events.json`). No se publican; aquí es donde se editan.
- `site/`: la app completa, lo que se publica (HTML, CSS y JavaScript sin dependencias de build).
  - `site/data/bundle.js`: el único archivo de datos que carga la app, generado desde `data/`.
  - `site/lugares/`, `site/artesanos/` y `site/cigarros/`: páginas estáticas por lugar, por artesano y por cigarro, generadas desde `data/`, igual que `site/sitemap.xml` y `site/robots.txt`.
  - `site/sw.js`: service worker (caché y modo offline).
  - `site/404.html`: página de "no encontrada".
  - `site/img/`, `site/icons/`, `site/fonts/`: imágenes, íconos y tipografías.
- `tools/`: los generadores (ver abajo). `tools/archive/` guarda scripts viejos de verificación que ya no se usan.
- `test/site.test.js`: pruebas (`node --test test/site.test.js`).
- `evidence/`: constancias de verificación de los cambios.

## Después de editar los JSON en `data/`

Desde la raíz del repositorio:

```bash
python3 tools/build_lugares.py
python3 tools/build_artesanos.py
python3 tools/rebuild_bundle.py
```

- `build_lugares.py`: `site/lugares/`, `site/sitemap.xml` y `site/robots.txt`.
- `build_artesanos.py`: `site/artesanos/`, `site/cigarros/` (llama a `build_cigarros.py`) y `site/sitemap.xml`.
- `rebuild_bundle.py`: `site/data/bundle.js`.

`tools/build_data.py` es distinto: crea los JSON de `data/` y el bundle a partir de un archivo de investigación nuevo (`python3 tools/build_data.py investigacion.json`).

## Publicación

- Hosting: Cloudflare Pages, proyecto `la-hoja-boricua`. Publica la carpeta `site/` desde la rama `main`.
- En cada publicación, Cloudflare corre los tres generadores de arriba, así que el sitio siempre sale de `data/`.
- Cada otra rama tiene su vista previa en `https://<rama>.la-hoja-boricua.pages.dev`.
- `www.lahojaboricua.com` redirige a `https://lahojaboricua.com`.
- GitHub Actions (`.github/workflows/test.yml`) corre las pruebas y los generadores en cada push y avisa si las páginas guardadas en git no coinciden con lo que producen los generadores.

## Probar localmente

cd site
python3 -m http.server 8080

function* executeRequest(request: Request) {
	const requestPath = new URL(request.url).pathname;

	// First, iterate through the routes (backwards) and execute "middlewares" on partial route matches
	for (const route of [...routes].reverse()) {
		if (route.method && route.method !== request.method) {
			continue;
		}

		// replaces with "\\$&", this prepends a backslash to the matched string, e.g. "[" becomes "\["
		const routeMatcher = match(route.routePath.replace(escapeRegex, "\\$&"), {
			end: false,
		});
		const mountMatcher = match(route.mountPath.replace(escapeRegex, "\\$&"), {
			end: false,
		});
		const matchResult = routeMatcher(requestPath);
		const mountMatchResult = mountMatcher(requestPath);
		if (matchResult && mountMatchResult) {
			for (const handler of route.middlewares.flat()) {
				yield {
					handler,
					params: matchResult.params as Params,
					path: mountMatchResult.path,
				};
			}
		}
	}

	// Then look for the first exact route match and execute its "modules"
	for (const route of routes) {
		if (route.method && route.method !== request.method) {
			continue;
		}
		const routeMatcher = match(route.routePath.replace(escapeRegex, "\\$&"), {
			end: true,
		});
		const mountMatcher = match(route.mountPath.replace(escapeRegex, "\\$&"), {
			end: false,
		});
		const matchResult = routeMatcher(requestPath);
		const mountMatchResult = mountMatcher(requestPath);
		if (matchResult && mountMatchResult && route.modules.length) {
			for (const handler of route.modules.flat()) {
				yield {
					handler,
					params: matchResult.params as Params,
					path: matchResult.path,
				};
			}
			break;
		}
	}
}
